\section{Delaunay Triangulation: Definition and Properties}
%======================================================================

Let $P = \{p_1, p_2, \ldots, p_n\} \subset \mathbb{R}^2$ be a finite set of points in the plane in
\emph{general position} (no three collinear, no four cocircular).

\begin{definition}[Triangulation]
A \emph{triangulation} $T$ of $P$ is a maximal planar straight-line graph whose vertex set is $P$.
Equivalently, $T$ partitions the convex hull $\mathrm{CH}(P)$ into triangles whose vertices are
elements of $P$ and whose interiors are pairwise disjoint.
\end{definition}

\begin{definition}[Delaunay Triangulation]
The \emph{Delaunay triangulation} $\mathcal{DT}(P)$ is the unique triangulation $T$ of $P$ such
that for every triangle $t \in T$, the open circumdisk $\mathrm{circ}(t)$ contains no point of $P$
in its interior:
\[
  \forall t = (p_i, p_j, p_k) \in T,\quad \forall p_\ell \in P \setminus \{p_i,p_j,p_k\}:\;
  p_\ell \notin \mathrm{int}\bigl(\mathrm{circ}(p_i,p_j,p_k)\bigr).
\]
\end{definition}

\begin{theorem}[Max-Min Angle]
Among all triangulations of $P$, the Delaunay triangulation maximizes the minimum interior angle.
Equivalently, it maximizes the vector of sorted angles in lexicographic order.
\end{theorem}

The Delaunay triangulation is dual to the \emph{Voronoi diagram} $\mathcal{V}(P)$: each Voronoi
cell is
\[
  V_i = \bigl\{x \in \mathbb{R}^2 : \|x - p_i\| \leq \|x - p_j\| \;\forall j \neq i\bigr\},
\]
and the dual edge connecting $p_i$ to $p_j$ in $\mathcal{DT}(P)$ corresponds to the Voronoi edge
shared by $V_i$ and $V_j$.

\section{The In-Circle Predicate (Shewchuk)}
%======================================================================

The cornerstone of Delaunay algorithms is an orientation-based \emph{in-circle test}.

\begin{definition}[In-Circle Predicate]
Given four points $A=(a_x,a_y)$, $B=(b_x,b_y)$, $C=(c_x,c_y)$, $D=(d_x,d_y)$ with $A,B,C$
ordered counterclockwise, define
\begin{equation}\label{eq:incircle}
  \mathrm{InCircle}(A,B,C,D) = \det \begin{pmatrix}
    a_x - d_x & a_y - d_y & (a_x-d_x)^2 + (a_y-d_y)^2 \\
    b_x - d_x & b_y - d_y & (b_x-d_x)^2 + (b_y-d_y)^2 \\
    c_x - d_x & c_y - d_y & (c_x-d_x)^2 + (c_y-d_y)^2
  \end{pmatrix}.
\end{equation}
\end{definition}

The sign of this determinant encodes the position of $D$ relative to the circumcircle of
$\triangle ABC$:
\begin{itemize}
  \item $\mathrm{InCircle}(A,B,C,D) > 0$: $D$ lies \emph{inside} the circumcircle $\Rightarrow$
        the diagonal must be \emph{flipped}.
  \item $\mathrm{InCircle}(A,B,C,D) = 0$: $A,B,C,D$ are \emph{cocircular}; either triangulation
        is locally Delaunay.
  \item $\mathrm{InCircle}(A,B,C,D) < 0$: $D$ lies \emph{outside} the circumcircle $\Rightarrow$
        no flip required.
\end{itemize}

\subsection{Derivation via the Lifting Map}

Map each point $p=(x,y)$ to a point $\hat{p} = (x,y,x^2+y^2)$ on the paraboloid
$z = x^2 + y^2$ in $\mathbb{R}^3$.  Four points are cocircular iff their lifted images are
\emph{coplanar}.  The signed volume of the tetrahedron $(\hat{A},\hat{B},\hat{C},\hat{D})$ is
\[
  \frac{1}{6}\det \begin{pmatrix}
    a_x & a_y & a_x^2+a_y^2 & 1 \\
    b_x & b_y & b_x^2+b_y^2 & 1 \\
    c_x & c_y & c_x^2+c_y^2 & 1 \\
    d_x & d_y & d_x^2+d_y^2 & 1
  \end{pmatrix},
\]
which equals $\tfrac{1}{6}\,\mathrm{InCircle}(A,B,C,D)$ after row-subtraction by the last row.
$D$ is below the plane of $\hat{A},\hat{B},\hat{C}$ iff it is inside the circumcircle
(positive determinant for CCW-ordered $A,B,C$).

\section{Guibas--Stolfi Quad-Edge Data Structure}
%======================================================================

\begin{definition}[Quad-Edge]
For every undirected edge $e = \{u,v\}$ in a planar subdivision, the
\emph{quad-edge} data structure stores four directed half-edges:
\[
  e,\quad e^{\mathrm{sym}},\quad e^{\mathrm{rot}},\quad e^{\mathrm{rot}^{-1}},
\]
where $e$ goes from $u$ to $v$, $e^{\mathrm{sym}}$ from $v$ to $u$, and $e^{\mathrm{rot}}$,
$e^{\mathrm{rot}^{-1}}$ are edges in the \emph{dual} graph.
\end{definition}

The fundamental navigation operators are:
\begin{align}
  \mathrm{Org}(e)  &= \text{origin vertex of } e, \\
  \mathrm{Dest}(e) &= \mathrm{Org}(e^{\mathrm{sym}}), \\
  \mathrm{Onext}(e) &= \text{next CCW edge with same origin}, \\
  \mathrm{Lnext}(e) &= e^{\mathrm{rot}^{-1}} \cdot \mathrm{Onext}(e^{\mathrm{rot}}),\quad
                       \text{(next CCW edge of same left face)}.
\end{align}

\subsection{Edge Flip in $O(1)$}

Suppose edge $e = (A,C)$ is the diagonal of quadrilateral $ABCD$ (where $B$ is to the left of $e$
and $D$ to the right).  The \emph{flip} replaces $e$ with $e' = (B,D)$:

\begin{enumerate}
  \item Detach $e$ from its origin/destination rings: update \texttt{Splice}$(e, \mathrm{Oprev}(e))$
        and \texttt{Splice}$(e^{\mathrm{sym}}, \mathrm{Oprev}(e^{\mathrm{sym}}))$.
  \item Set $\mathrm{Org}(e) \leftarrow B$, $\mathrm{Dest}(e) \leftarrow D$.
  \item Reattach: \texttt{Splice}$(e, \mathrm{Lnext}(a))$ and
        \texttt{Splice}$(e^{\mathrm{sym}}, \mathrm{Lnext}(c))$.
\end{enumerate}

Total pointer updates: $O(1)$.  The operation preserves the Euler formula $V - E + F = 2$.

\section{Incremental Insertion Algorithm}
%======================================================================

The standard incremental algorithm for $\mathcal{DT}(P)$ inserts points one at a time:

\begin{enumerate}
  \item \textbf{Locate}: Walk from a seed triangle using the orientation predicate
        \[
          \mathrm{Orient}(A,B,C) = \det\begin{pmatrix}
            a_x-c_x & a_y-c_y \\ b_x-c_x & b_y-c_y
          \end{pmatrix} \begin{cases} >0 & \text{CCW} \\ =0 & \text{collinear} \\ <0 & \text{CW}\end{cases}
        \]
        to locate the triangle $\triangle ABC$ containing the new point $p$.
  \item \textbf{Split}: Connect $p$ to $A$, $B$, $C$, replacing one triangle with three.
  \item \textbf{Legalize}: For each new edge $e_i = (p, v_i)$ with opposite vertex $w_i$,
        if $\mathrm{InCircle}(p, v_i, w_i, \text{fourth vertex}) > 0$, flip $e_i$ and recurse.
\end{enumerate}

\begin{theorem}[Correctness and Complexity]
The incremental algorithm produces the unique Delaunay triangulation of $P$ in expected time
$O(n \log n)$ for random insertion order, and worst-case $O(n^2)$ for adversarial order.
\end{theorem}

The expected time analysis follows from the observation that each point is re-examined only
$O(1)$ times on average during the legalization phase, under a random permutation of insertions.

\section{Floating-Point Robustness}
%======================================================================

The key vulnerability of the incremental algorithm is the in-circle test near the zero boundary.

\subsection{The Cocircular Problem}

For four points lying on a common circle (or nearly so):
\begin{itemize}
  \item \textbf{Exact arithmetic}: $\mathrm{InCircle}(A,B,C,D) = 0$ — both diagonals are valid.
  \item \textbf{IEEE 754 double}: roundoff error gives $\mathrm{InCircle}(A,B,C,D) = \pm\varepsilon$
        for some small $\varepsilon \neq 0$.
  \item \textbf{Consequence}: the algorithm may flip the same edge repeatedly, producing an
        infinite loop or an inconsistent triangulation with missing triangles.
\end{itemize}

\subsection{Adaptive Tolerance Fix}

The fix applied in \texttt{cv::Subdiv2D} uses an adaptive tolerance $\delta$ based on the
magnitude of the input coordinates:
\begin{equation}\label{eq:delta}
  \delta = \varepsilon_{\mathrm{machine}} \cdot
  \bigl(|a_x|+|a_y|+|b_x|+|b_y|+|c_x|+|c_y|+|d_x|+|d_y|\bigr)^3,
\end{equation}
where $\varepsilon_{\mathrm{machine}} \approx 2.22 \times 10^{-16}$ for IEEE 754 double precision.

The modified predicate becomes:
\[
  \tilde{\mathrm{InCircle}}(A,B,C,D) = \begin{cases}
    +1 & \text{if } \mathrm{InCircle}(A,B,C,D) > +\delta \\
     0 & \text{if } |\mathrm{InCircle}(A,B,C,D)| \leq \delta \quad\text{(treat as cocircular)}\\
    -1 & \text{if } \mathrm{InCircle}(A,B,C,D) < -\delta
  \end{cases}
\]

\begin{remark}
Shewchuk (1996) provides a more sophisticated \emph{adaptive exact arithmetic} approach that
computes the determinant to only as many bits of precision as needed to determine the sign,
avoiding both false positives and unnecessary precision overhead.
\end{remark}

\section{Voronoi Duality}
%======================================================================

The Voronoi diagram $\mathcal{V}(P)$ partitions $\mathbb{R}^2$ into cells
\[
  V_i = \bigl\{x \in \mathbb{R}^2 : \|x - p_i\| \leq \|x - p_j\|\;\forall j \neq i\bigr\}.
\]
The \emph{duality} between $\mathcal{DT}(P)$ and $\mathcal{V}(P)$ is exact:

\begin{center}
\begin{tabular}{ll}
\textbf{Delaunay} & \textbf{Voronoi} \\
\hline
Vertex $p_i$ & Cell $V_i$ \\
Edge $(p_i,p_j)$ & Edge between $V_i$ and $V_j$ \\
Triangle $(p_i,p_j,p_k)$ & Voronoi vertex (circumcenter of $p_i,p_j,p_k$) \\
\end{tabular}
\end{center}

In OpenCV's \texttt{Subdiv2D}, both structures are maintained simultaneously via the quad-edge
representation: primal edges encode $\mathcal{DT}(P)$ and their rotations encode $\mathcal{V}(P)$.

\section{Complexity Analysis Summary}
%======================================================================

\begin{center}
\begin{tabular}{lll}
\hline
\textbf{Operation} & \textbf{Expected} & \textbf{Worst case} \\
\hline
Point location (walk) & $O(\sqrt{n})$ & $O(n)$ \\
Single insertion + legalization & $O(\log n)$ & $O(n)$ \\
Total incremental algorithm & $O(n \log n)$ & $O(n^2)$ \\
Edge flip & $O(1)$ & $O(1)$ \\
\hline
\end{tabular}
\end{center}

With a Kirkpatrick point-location structure, the walk step can be replaced by $O(\log n)$
deterministic point location, achieving $O(n \log n)$ worst-case total.
