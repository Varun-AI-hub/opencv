% ----------------------------------------------------------------
% Section 3: In-circle test determinant
% ----------------------------------------------------------------

\begin{scope}[yshift=-4cm]

    \node[font=\normalsize\bfseries] at (5, 0.3)
        {In-Circle Test: Is $D$ inside circumcircle of $\triangle ABC$?};

    % Formula box
    \node[draw, rounded corners, fill=yellow!15, align=left,
          font=\footnotesize, inner sep=8pt] at (5, -1.0)
    {
        $\text{det} = \begin{vmatrix}
            A_x - D_x & A_y - D_y & (A_x^2 + A_y^2) - (D_x^2 + D_y^2) \\
            B_x - D_x & B_y - D_y & (B_x^2 + B_y^2) - (D_x^2 + D_y^2) \\
            C_x - D_x & C_y - D_y & (C_x^2 + C_y^2) - (D_x^2 + D_y^2)
        \end{vmatrix}$ \\[6pt]
        \hspace{1em}$> 0$: $D$ inside circle $\Rightarrow$ flip diagonal \\
        \hspace{1em}$= 0$: $D$ on circle (degenerate) \\
        \hspace{1em}$< 0$: $D$ outside circle $\Rightarrow$ keep diagonal
    };

    % Floating-point problem
    \node[draw, rounded corners, fill=red!10, align=left,
          font=\footnotesize, inner sep=8pt] at (5, -3.5)
    {
        \textbf{Floating-point problem:} \\
        Theory: $\text{det} = 0.000000000000000$ (exact zero) \\
        Reality: $\text{det} = +0.000000000000002$ (flip!) \\
        \hspace{4.5em}or: $\text{det} = -0.000000000000001$ (don't flip) \\[4pt]
        $\Rightarrow$ Non-deterministic result depending on rounding
    };

\end{scope}

    \node[draw, rounded corners, fill=green!10, align=left,
          font=\footnotesize, inner sep=8pt] at (5, -2.5)
    {
        \textbf{Before fix:} \texttt{if (det > 0) \{ flip(); \}} \\
        \textbf{After fix:} \texttt{if (det > epsilon) \{ flip(); \}} \\[4pt]
        Degenerate cases (all 4 points on one circle) now produce a\\
        stable, consistent triangulation instead of random behavior.
    };

\end{scope}
